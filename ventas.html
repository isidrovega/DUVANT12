create or replace function public.delete_sale( p_sale_id uuid ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user_id uuid; v_profile public.profiles%rowtype; v_sale public.sales%rowtype; v_restored_quantity integer; begin /* ===================================== USUARIO AUTENTICADO ===================================== */ v_user_id := auth.uid(); if v_user_id is null then raise exception 'Debes iniciar sesión.'; end if; /* ===================================== VALIDAR ADMINISTRADOR ===================================== */ select * into v_profile from public.profiles where id = v_user_id; if not found then raise exception 'No se encontró el perfil del usuario.'; end if; if v_profile.active is not true then raise exception 'Tu cuenta está desactivada.'; end if; if v_profile.role <> 'admin' then raise exception 'Solo los administradores pueden eliminar ventas.'; end if; /* ===================================== OBTENER Y BLOQUEAR LA VENTA ===================================== */ select * into v_sale from public.sales where id = p_sale_id for update; if not found then raise exception 'La venta no existe o ya fue eliminada.'; end if; /* ===================================== DEVOLVER UNIDAD AL INVENTARIO ===================================== */ update public.perfumes set quantity = quantity + 1, updated_at = now() where id = v_sale.perfume_id returning quantity into v_restored_quantity; if not found then raise exception 'No se encontró el perfume asociado a esta venta.'; end if; /* ===================================== ELIMINAR VENTA ===================================== */ delete from public.sales where id = p_sale_id; /* ===================================== RESPUESTA ===================================== */ return jsonb_build_object( 'success', true, 'saleId', v_sale.id, 'perfumeId', v_sale.perfume_id, 'perfumeName', v_sale.perfume_name, 'restoredQuantity', v_restored_quantity ); end; $$; /* ===================================== PERMISOS ===================================== */ revoke all on function public.delete_sale(uuid) from public; grant execute on function public.delete_sale(uuid) to authenticated;
